    function rand(min, max){ return Math.floor(Math.random()*(max-min+1))+min; }


    function makeTerminal(x, y, title="C:\\\\Windows\\\\System32\\\\cmd.exe"){

      const el = document.createElement('div');

      el.className = "term fadeIn";

      el.style.left = x + "px";

      el.style.top  = y + "px";

      el.innerHTML = `

        <div class="bar">

          <div>${title}</div>

          <div style="width:42px"></div>

        </div>

        <pre></pre>

      `;

      desktop.appendChild(el);

      return el;

    }


    function typeInto(pre, text, speed=10){

      let i = 0;

      const timer = setInterval(() => {

        pre.textContent += text[i++] || "";

        if(i >= text.length) clearInterval(timer);

        pre.scrollTop = pre.scrollHeight;

      }, speed);

    }


    function terminalScript(){

      const lines = [

        "Microsoft Windows [Version 10.0.19045.0]",

        "(c) Microsoft Corporation. All rights reserved.",

        "",

        "C:\\\\> love.exe --target=valentine --mode=maximum",

        "Loading feelings... OK",

        "Compiling hugs... OK",

        "Linking kisses... OK",

        "",

        BIG_HEART,

        "",

        "SUCCESS: You are my Valentine 💘",

        fromName

      ];

      return lines.join("\\n");

    }


    function spawnFakeCmdStorm(count=10){

      const w = window.innerWidth, h = window.innerHeight;

      for(let i=0;i<count;i++){

        const x = rand(10, Math.max(10, w-380));

        const y = rand(10, Math.max(10, h-240));

        const t = makeTerminal(x, y);

        const pre = t.querySelector('pre');

        // stagger

        setTimeout(() => typeInto(pre, terminalScript(), rand(4, 14)), i * 120);

      }

    }


    function floatHearts(durationMs=6000){

      const emojis = ["💗","💖","💘","💝","💕","❤️"];

      const start = Date.now();

      const interval = setInterval(() => {

        const s = document.createElement("div");

        s.className = "float-heart";

        s.textContent = emojis[rand(0, emojis.length-1)];

        s.style.left = rand(0, window.innerWidth) + "px";

        s.style.setProperty("--dx", rand(-120, 120) + "px");

        s.style.animationDuration = rand(3500, 7000) + "ms";

        document.body.appendChild(s);

        setTimeout(()=>s.remove(), 8000);

        if(Date.now() - start > durationMs) clearInterval(interval);

      }, 90);

    }
